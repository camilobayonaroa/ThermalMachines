% NB15b -- Cinematica de turbomaquinas axiales: (a) cambio volumetrico a lo largo del canal anular
% (A = m/(rho Va) -> el anulo se cierra en el compresor y se abre en la turbina); (b) etapa de compresor
% rotor-estator; (c) etapa de turbina estator-rotor, con triangulos de velocidad C = W + U y ecuacion de Euler.
% Reemplaza las figuras de triangulos (rotor/estator) de la entrega PlantaTermoelectrica (Sanchez, Aparicio, Roa).

\definecolor{caliente}{HTML}{E34948}
\begin{tikzpicture}[x=1cm,y=1cm,
  C/.style={-{Stealth[length=2.4mm]}, line width=1.2pt, cabs},
  W/.style={-{Stealth[length=2.4mm]}, line width=1.2pt, wrel},
  U/.style={-{Stealth[length=2.4mm]}, line width=1.2pt, uvel},
  alabe/.style={line width=3pt, line cap=round, tinta},
  tit/.style={font=\small\bfseries, text=tinta}]
  % ================= (a) canal anular: cambio volumetrico =================
  \begin{scope}[shift={(0,9.2)}]
    \node[tit, anchor=west] at (-0.3,3.1) {(a) Cambio volum\'etrico en el canal anular: $\dot m=\rho\,V_a\,A$,\quad $A_i=\pi\,(r_{tip,i}^2-r_{hub,i}^2)$};
    % eje
    \draw[tinta, dash dot, line width=0.5pt] (-0.3,0) -- (17.3,0) node[right, font=\scriptsize] {eje};
    % compresor: carcasa baja y cubo sube (anulo se cierra)
    \fill[frio!12] (0,0.9) -- (6,1.6) -- (6,2.0) -- (0,2.6) -- cycle;
    \draw[tinta, line width=1pt] (0,2.6) -- (6,2.0); \draw[tinta, line width=1pt] (0,0.9) -- (6,1.6);
    \foreach \i in {0,...,14} { \pgfmathsetmacro{\x}{0.2+\i*0.39}
      \pgfmathsetmacro{\yb}{0.9+0.7*\x/6} \pgfmathsetmacro{\yt}{2.6-0.6*\x/6}
      \draw[frio!70!tinta, line width=0.9pt] (\x,\yb) -- (\x,\yt);}
    \node[lbl, font=\footnotesize, align=center] at (3,0.45) {compresor: $\rho\uparrow$ (15 etapas) $\Rightarrow A\downarrow$};
    % camara
    \draw[tinta, line width=1pt, fill=caliente!25, rounded corners=3pt] (6.3,1.5) rectangle (9.0,2.2);
    \node[lbl, font=\footnotesize] at (7.65,1.85) {combusti\'on};
    \node[lbl, font=\scriptsize, align=center] at (7.65,1.0) {$T\uparrow$ a $p\approx$ cte\\$\Rightarrow\rho\downarrow$};
    % turbina: anulo se abre
    \fill[caliente!12] (9.3,1.5) -- (13.3,0.6) -- (13.3,3.0) -- (9.3,2.1) -- cycle;
    \draw[tinta, line width=1pt] (9.3,2.1) -- (13.3,3.0); \draw[tinta, line width=1pt] (9.3,1.5) -- (13.3,0.6);
    \foreach \x in {9.9,11.2,12.6} { \pgfmathsetmacro{\yb}{1.5-0.9*(\x-9.3)/4} \pgfmathsetmacro{\yt}{2.1+0.9*(\x-9.3)/4}
      \draw[caliente!80!tinta, line width=1.6pt] (\x,\yb) -- (\x,\yt);}
    \node[lbl, font=\footnotesize, align=center] at (11.3,0.2) {turbina: $\rho\downarrow$ (3 etapas) $\Rightarrow A\uparrow$};
    % cotas r_tip, r_hub
    \draw[cota] (0.05,0) -- (0.05,2.6); \node[lbl, font=\scriptsize, left] at (0.05,1.9) {$r_{tip}$};
    \draw[cota] (-0.2,0) -- (-0.2,0.9); \node[lbl, font=\scriptsize, left] at (-0.2,0.45) {$r_{hub}$};
    % relacion
    \node[draw=tinta, rounded corners=2pt, fill=white, font=\footnotesize, align=left, anchor=west] at (13.9,1.6)
      {$\dfrac{A_{i+1}}{A_i}=\dfrac{\rho_i\,V_{a,i}}{\rho_{i+1}\,V_{a,i+1}}$\\[2pt]con $V_a\approx$ cte:\\$A\propto 1/\rho\propto T/p$};
  \end{scope}
  % ================= (b) etapa de compresor: rotor + estator =================
  \begin{scope}[shift={(0,0)}]
    \node[tit, anchor=west] at (-0.3,7.6) {(b) Etapa de compresor: el rotor \emph{a\~nade} momento angular};
    \draw[tinta, line width=0.6pt] (0.6,1.2) -- (0.6,7.0); \draw[tinta, line width=0.6pt] (2.4,1.2) -- (2.4,7.0);
    \draw[tinta, line width=0.6pt] (3.4,1.2) -- (3.4,7.0); \draw[tinta, line width=0.6pt] (5.2,1.2) -- (5.2,7.0);
    \foreach \y in {2.9,4.4,5.9} {
      \draw[alabe] (0.9,\y) to[out=-54, in=139] (2.1,\y-1.25);     % rotor: W1 -> W2 (giro hacia axial)
      \draw[alabe] (3.7,\y-0.6) to[out=37, in=194] (4.9,\y);      % estator: C2 -> C3
    }
    \node[lbl, font=\footnotesize\bfseries] at (1.5,0.85) {rotor}; \node[lbl, font=\footnotesize\bfseries] at (4.3,0.85) {estator};
    \draw[-{Stealth[length=2.6mm]}, uvel, line width=1.2pt] (1.5,6.35) -- (1.5,7.15) node[right, font=\scriptsize] {$U=\omega r_m$};
    \foreach \x/\n in {0.3/1, 2.9/2, 5.5/3} \node[circle, draw=tinta, fill=white, inner sep=1pt, font=\scriptsize\bfseries] at (\x,7.0) {\n};
    % triangulos (origen comun) -- x axial, y tangencial
    \begin{scope}[shift={(6.6,4.3)}, scale=0.95]
      \coordinate (O) at (0,0);
      \draw[C] (O) -- (1.6,0.4) node[pos=0.55, above, font=\footnotesize] {$C_1$};
      \draw[W] (O) -- (1.6,-2.2) node[pos=0.55, left, font=\footnotesize] {$W_1$};
      \draw[U] (1.6,-2.2) -- (1.6,0.4) node[pos=0.5, right, font=\footnotesize] {$U$};
      \begin{scope}[shift={(2.3,0)}]
      \draw[C] (0,0) -- (1.6,1.2); \node[cabs, font=\footnotesize] at (0.55,0.85) {$C_2$};
      \draw[W] (0,0) -- (1.6,-1.4); \node[wrel, font=\footnotesize] at (0.55,-0.95) {$W_2$};
      \draw[U] (1.6,-1.4) -- (1.6,1.2) node[pos=0.35, right, font=\footnotesize] {$U$};
      \end{scope}
      \node[lbl, font=\scriptsize] at (0.8,-2.55) {entrada rotor (1)}; \node[lbl, font=\scriptsize] at (3.1,-2.55) {salida rotor (2)};
      \draw[tinta, line width=0.5pt, densely dotted] (1.6,0.4) -- (4.5,0.4); \draw[tinta, line width=0.5pt, densely dotted] (3.9,1.2) -- (4.5,1.2);
      \draw[cota] (4.5,0.4) -- (4.5,1.2); \node[lbl, font=\scriptsize, right] at (4.5,0.8) {$\Delta C_u$};
      \draw[tinta, line width=0.5pt] (0.7,0) arc (0:-54:0.7); \node[lbl, font=\scriptsize] at (0.55,-0.95) {$\beta_1$};
    \end{scope}
    \node[draw=tinta, rounded corners=2pt, fill=white, font=\footnotesize, align=left, anchor=north west] at (-0.3,0.45)
      {Euler: $\Delta h_0=U\,(C_{u2}-C_{u1})>0$,\quad $\tau=\dot m\,r_m\,(C_{u2}-C_{u1})$,\quad $\dot W=\tau\,\omega$\\
       difusi\'on en $W$ ($|W_2|<|W_1|$) y en $C$ (estator) $\Rightarrow p\uparrow$\\ l\'imite de desprendimiento: $W_2/W_1\gtrsim0.72$ (de Haller) $\Rightarrow$ muchas etapas};
  \end{scope}
  % ================= (c) etapa de turbina: estator + rotor =================
  \begin{scope}[shift={(11.2,0)}]
    \node[tit, anchor=west] at (-0.3,7.6) {(c) Etapa de turbina: el rotor \emph{extrae} momento angular};
    \draw[tinta, line width=0.6pt] (0.6,1.2) -- (0.6,7.0); \draw[tinta, line width=0.6pt] (2.4,1.2) -- (2.4,7.0);
    \draw[tinta, line width=0.6pt] (3.4,1.2) -- (3.4,7.0); \draw[tinta, line width=0.6pt] (5.2,1.2) -- (5.2,7.0);

    \draw[-{Stealth[length=2.6mm]}, uvel, line width=1.2pt] (4.3,6.35) -- (4.3,7.15) node[right, font=\scriptsize] {$U$};
    \foreach \x/\n in {0.3/1, 2.9/2, 5.5/3} \node[circle, draw=tinta, fill=white, inner sep=1pt, font=\scriptsize\bfseries] at (\x,7.0) {\n};
    \begin{scope}[shift={(6.4,3.1)}, scale=0.85]
      \coordinate (O) at (0,0);
      \draw[C] (O) -- (1.6,3.2) node[pos=0.6, left, font=\footnotesize] {$C_2$};
      \draw[W] (O) -- (1.6,1.0) node[pos=0.6, below right, font=\footnotesize] {$W_2$};
      \draw[U] (1.6,1.0) -- (1.6,3.2) node[pos=0.5, right, font=\footnotesize] {$U$};
      \begin{scope}[shift={(2.3,0)}]

      \end{scope}

      \draw[cota] (4.6,-0.2) -- (4.6,3.2); \node[lbl, font=\scriptsize, right] at (4.6,1.5) {$\Delta C_u$};
      \node[lbl, font=\scriptsize] at (0.6,-2.9) {salida estator (2)}; \node[lbl, font=\scriptsize] at (3.6,-2.9) {salida rotor (3)};
    \end{scope}
    \node[draw=tinta, rounded corners=2pt, fill=white, font=\footnotesize, align=left, anchor=north west] at (-0.3,0.45)
      {Euler: $\Delta h_0=U\,(C_{u2}-C_{u3})>0$ extra\'ido del gas\\ grado de reacci\'on $R=\dfrac{\Delta h_{rotor}}{\Delta h_{etapa}}$\\
       aceleraci\'on (no difusi\'on) $\Rightarrow$ gran deflexi\'on admisible: pocas etapas (3 vs. 15)};
  \end{scope}
